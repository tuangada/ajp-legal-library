/* --------------------------
  App constants & helpers
---------------------------*/
const STORAGE_KEY = "legalDocsV4";
const PW_KEY = "legalDocsV4_pw"; // client-side password (optional)
let docs = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");
let fuse = null;
let currentPage = 1;
const pageSize = 8; // items per page for pagination
let lastFilterCategory = "";

function save(){
  localStorage.setItem(STORAGE_KEY, JSON.stringify(docs));
}
function uid(){ return Date.now().toString(36) + Math.random().toString(36).slice(2,8); }
function normalize(text){ return (text||"").toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g,""); }

/* --------------------------
  Initialize Fuse (fuzzy search)
---------------------------*/
function initFuse(){
  const options = {
    includeScore: true,
    threshold: 0.35, // 0.0 = exact, 1.0 = very fuzzy. adjust if needed
    keys: [
      {name:"title", weight:0.6},
      {name:"category", weight:0.25},
      {name:"note", weight:0.15}
    ]
  };
  fuse = new Fuse(docs, options);
}

/* --------------------------
  Render functions
---------------------------*/
function renderList(q=""){
  const listEl = document.getElementById("list");
  listEl.innerHTML = "";

  // Search via Fuse when there's query, else take all
  let result = [];
  if(q && q.trim().length>0){
    const fuseRes = fuse.search(q);
    result = fuseRes.map(r => r.item);
  } else {
    result = [...docs];
  }

  // Apply category filter
  if(lastFilterCategory && lastFilterCategory.trim()!=""){
    const catNorm = normalize(lastFilterCategory);
    result = result.filter(d => normalize(d.category).includes(catNorm));
  }

  // Sorting
  const sortVal = document.getElementById("sortSel").value;
  if(sortVal==="new") result.sort((a,b)=>b.createdAt - a.createdAt);
  else if(sortVal==="old") result.sort((a,b)=>a.createdAt - b.createdAt);
  else if(sortVal==="az") result.sort((a,b)=>a.title.localeCompare(b.title));
  else if(sortVal==="za") result.sort((a,b)=>b.title.localeCompare(a.title));

  // Pagination
  const total = result.length;
  const pages = Math.max(1, Math.ceil(total / pageSize));
  if(currentPage > pages) currentPage = pages;
  const start = (currentPage-1)*pageSize;
  const pageItems = result.slice(start, start+pageSize);

  // Render items
  pageItems.forEach(d => {
    const div = document.createElement("div");
    div.className = "doc";
    div.innerHTML = `
      <div class="left">
        <h3>${escapeHtml(d.title)}</h3>
        <div class="meta"><span class="tag">${escapeHtml(d.category)}</span> &nbsp; • &nbsp;
            ${new Date(d.createdAt).toLocaleString()}</div>
        <p>${escapeHtml(d.note)}</p>
      </div>
      <div class="actions">
        <button class="small" onclick="viewDoc('${d.id}')">Lihat</button>
        <button class="small" onclick="openEdit('${d.id}')">Edit</button>
        <button class="small" onclick="confirmDelete('${d.id}')">Hapus</button>
      </div>
    `;
    listEl.appendChild(div);
  });

  // meta
  document.getElementById("metaCount").textContent = `${total} dokumen (halaman ${currentPage}/${pages})`;

  // pagination controls
  renderPagination(pages);
}

/* escape to avoid HTML injection */
function escapeHtml(s){ if(!s) return ""; return s.replaceAll("&","&amp;").replaceAll("<","&lt;").replaceAll(">","&gt;"); }

function renderPagination(pages){
  const el = document.getElementById("pagination");
  el.innerHTML = "";
  for(let i=1;i<=pages;i++){
    const btn = document.createElement("button");
    btn.className = "page" + (i===currentPage ? " active" : "");
    btn.textContent = i;
    btn.onclick = ()=>{ currentPage = i; renderList(document.getElementById("q").value.trim()); };
    el.appendChild(btn);
  }
}

/* --------------------------
  CRUD operations
---------------------------*/
function addDoc(obj){
  const d = {
    id: uid(),
    title: obj.title || "Tanpa Judul",
    category: obj.category || "Umum",
    note: obj.note || "",
    createdAt: Date.now(),
    updatedAt: Date.now()
  };
  docs.push(d);
  save(); initFuse(); renderList(document.getElementById("q").value.trim());
}

function updateDoc(id, payload){
  const idx = docs.findIndex(x=>x.id===id);
  if(idx>=0){
    docs[idx] = {...docs[idx], ...payload, updatedAt: Date.now()};
    save(); initFuse(); renderList(document.getElementById("q").value.trim());
  }
}
function removeDoc(id){
  docs = docs.filter(d=>d.id!==id);
  save(); initFuse(); renderList(document.getElementById("q").value.trim());
}

document.getElementById("btnBackup").addEventListener("click", ()=>{

document.getElementById("btnRestore").addEventListener("click", ()=>{
  document.getElementById("fileInput").click();
});
document.getElementById("fileInput").addEventListener("change", function(e){
  const file = e.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = function(ev){
    try{
      const parsed = JSON.parse(ev.target.result);
      if(!Array.isArray(parsed)) throw new Error("Format tidak sesuai (harus array)");
      if(!confirm(`Restore akan menimpa data saat ini dengan ${parsed.length} item. Lanjutkan?`)) return;
      docs = parsed.map(it=>{
        // normalize older items (ensure id and timestamps)
        return {
          id: it.id || uid(),
          title: it.title || "",
          category: it.category || "",
          note: it.note || "",
          createdAt: it.createdAt || Date.now(),
          updatedAt: it.updatedAt || Date.now()
        };
      });
      save(); initFuse(); renderList(document.getElementById("q").value.trim());
      alert("Restore berhasil");
    }catch(err){
      alert("Gagal restore: " + err.message);
    }
  };
  reader.readAsText(file);
  e.target.value = "";
});

/* Export CSV */
document.getElementById("btnExportCsv").addEventListener("click", ()=>{
  if(!docs.length) return alert("Tidak ada data untuk diexport");
  const headers = ["id","title","category","note","createdAt","updatedAt"];
  const rows = docs.map(d => headers.map(h => `"${(d[h]||"").toString().replace(/"/g,'""')}"`).join(","));
  const csv = [headers.join(","), ...rows].join("\n");
  const blob = new Blob([csv], {type:"text/csv"});
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a"); a.href = url; a.download = "legal_docs_export.csv"; a.click(); URL.revokeObjectURL(url);
});

/* Export PDF (basic) using jsPDF */
document.getElementById("btnExportPdf").addEventListener("click", async ()=>{
  if(!docs.length) return alert("Tidak ada data untuk diexport");
  const { jsPDF } = window.jspdf;
  const docPdf = new jsPDF({unit:"pt", format:"a4"});
  docPdf.setFontSize(12);
  docPdf.setTextColor(20,20,20);
  let y = 40;
  docPdf.setFontSize(16);
  docPdf.text("Legal Document Library - Export", 40, y); y+=24;
  docPdf.setFontSize(10);
  docs.forEach((d, idx)=>{
    const title = `${idx+1}. ${d.title} [${d.category}]`;
    docPdf.text(title, 40, y); y+=16;
    const noteLines = docPdf.splitTextToSize(d.note || "-", 500);
    noteLines.forEach(ln => { docPdf.text(ln, 60, y); y+=12; });
    y+=10;
    if(y > 700){ docPdf.addPage(); y = 40; }
  });
  docPdf.save("legal_docs_export.pdf");
});

    alert("Password tersimpan di browser. Fitur ini hanya memberikan proteksi ringan (client-side).");
  } else {
    alert("Password harus >= 4 karakter. Dibatalkan.");
  }
});

/* On load: check password */
function checkPasswordOnLoad(){
  const pw = localStorage.getItem(PW_KEY);
  if(!pw) return Promise.resolve(true);
  return new Promise((res)=>{
    let tries = 0;
    function ask(){
      const v = prompt("Sistem terkunci. Masukkan password:");
      if(v === pw){ res(true); }
      else {
        tries++;
        if(tries>=3){ alert("Terlalu banyak percobaan. Aplikasi akan tetap berjalan tetapi data mungkin tetap terlihat."); res(true); }
        else { alert("Password salah."); ask(); }
      }
    }
    ask();
  });
}
